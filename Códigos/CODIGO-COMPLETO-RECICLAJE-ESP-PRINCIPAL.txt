int get_char_index(char c) {
    if (c < 32 || c > 126) c = 32;
    return c - 32;
}

void mover_servomotor(ledc_channel_t channel, int angulo) {
    uint32_t duty;
    
    if (angulo < 0) angulo = 0;
    if (angulo > 180) angulo = 180;
    
    duty = 409 + (angulo * 410) / 180;
    
    ESP_LOGI(TAG, "Servo canal %d -> %d° (duty: %lu)", channel, angulo, duty);
    
    ESP_ERROR_CHECK(ledc_set_duty(LEDC_LOW_SPEED_MODE, channel, duty));
    ESP_ERROR_CHECK(ledc_update_duty(LEDC_LOW_SPEED_MODE, channel));
}

void controlar_servo2_objeto_detectado() {
    if (!servo2_activado) {
        mover_servomotor(LEDC_CHANNEL_SERVO2, 180);
        servo2_activado = true;
        servo2_tiempo_activacion = xTaskGetTickCount();
        servo2_completado = false;
        ESP_LOGI(TAG, "Servo2 ACTIVADO por objeto detectado");
    }
}

void controlar_servo_por_objeto(const char* objeto) {
    ESP_LOGI(TAG, "Procesando objeto: %s", objeto);
    
    // Configurar texto para display según el objeto detectado
    if (strstr(objeto, "Botella") != NULL || strstr(objeto, "Otros_plasticos") != NULL) {
        strcpy(objeto_display, "PLASTICO");
        mover_servomotor(LEDC_CHANNEL_SERVO3, 180);
        servo3_activado = true;
        servo_nuevo_tiempo_activacion = xTaskGetTickCount();
        ESP_LOGI(TAG, "Servo3 activado para plásticos");
    }
    else if (strstr(objeto, "Papel") != NULL || strstr(objeto, "Caja") != NULL || 
             strstr(objeto, "Otros_carton") != NULL) {
        strcpy(objeto_display, "PAPEL");
        mover_servomotor(LEDC_CHANNEL_SERVO4, 180);
        servo4_activado = true;
        servo_nuevo_tiempo_activacion = xTaskGetTickCount();
        ESP_LOGI(TAG, "Servo4 activado para papel/cartón");
    }
    else if (strstr(objeto, "Lata") != NULL || strstr(objeto, "Otros_aluminio") != NULL) {
        strcpy(objeto_display, "METAL");
        mover_servomotor(LEDC_CHANNEL_SERVO5, 180);
        servo5_activado = true;
        servo_nuevo_tiempo_activacion = xTaskGetTickCount();
        ESP_LOGI(TAG, "Servo5 activado para metales");
    }
    else if (strstr(objeto, "No_reciclable") != NULL) {
        strcpy(objeto_display, "NO RECICLABLE");
        ESP_LOGI(TAG, "Objeto no reciclable - servos quietos");
    }
    else {
        // Por si llega algún objeto no reconocido
        strncpy(objeto_display, objeto, sizeof(objeto_display)-1);
        objeto_display[sizeof(objeto_display)-1] = '\0';
    }
    
    // Activar display del objeto
    mostrar_objeto = true;
    tiempo_display_objeto = xTaskGetTickCount();
    
    // Siempre activar Servo2 cuando se detecta cualquier objeto
    controlar_servo2_objeto_detectado();
}

// Callbacks ESP-NOW
void espnow_receive_cb(const esp_now_recv_info_t *recv_info, const uint8_t *data, int len) {
    if (len == sizeof(espnow_data_t)) {
        memcpy(&received_data, data, sizeof(espnow_data_t));
        
        ESP_LOGI(TAG, "Objeto detectado: %s (confianza: %.2f)", 
                 received_data.objeto_detectado, received_data.confidence);
        
        strcpy(objeto_actual, received_data.objeto_detectado);
        objeto_detectado_espnow = true;
        
        // Controlar servos según objeto detectado
        controlar_servo_por_objeto(received_data.objeto_detectado);
        
        // El display se actualizará automáticamente en update_ultrasonic_display()
    }
}

    ESP_LOGI(TAG, "Estado envio ESP-NOW: %s", status == ESP_NOW_SEND_SUCCESS ? "OK" : "Fallo");
}

// Inicialización WiFi y ESP-NOW
void wifi_init() {
    ESP_ERROR_CHECK(esp_netif_init());
    ESP_ERROR_CHECK(esp_event_loop_create_default());
    
    wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
    ESP_ERROR_CHECK(esp_wifi_init(&cfg));
    ESP_ERROR_CHECK(esp_wifi_set_storage(WIFI_STORAGE_RAM));
    ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_STA));
    ESP_ERROR_CHECK(esp_wifi_start());
}

void setup_espnow() {
    wifi_init();
    
    ESP_ERROR_CHECK(esp_now_init());
    
    ESP_ERROR_CHECK(esp_now_register_recv_cb(espnow_receive_cb));
    ESP_ERROR_CHECK(esp_now_register_send_cb(espnow_send_status_cb));
    
    esp_now_peer_info_t peerInfo = {};
    memcpy(peerInfo.peer_addr, peerMac, 6);
    peerInfo.channel = 0;
    peerInfo.encrypt = false;
    
    ESP_ERROR_CHECK(esp_now_add_peer(&peerInfo));
    
    ESP_LOGI(TAG, "ESP-NOW configurado. Peer: %02X:%02X:%02X:%02X:%02X:%02X", 
             peerMac[0], peerMac[1], peerMac[2], peerMac[3], peerMac[4], peerMac[5]);
}

// Configuración servomotores
void setup_servomotores() {
    ledc_timer_config_t timer_config = {
        .speed_mode = LEDC_LOW_SPEED_MODE,
        .duty_resolution = LEDC_TIMER_13_BIT,
        .timer_num = LEDC_TIMER_0,
        .freq_hz = 50,
        .clk_cfg = LEDC_AUTO_CLK,
        .deconfigure = false
    };
    ESP_ERROR_CHECK(ledc_timer_config(&timer_config));

    ledc_channel_config_t channels[5];
    
    // Inicializar todos los canales con memset
    for (int i = 0; i < 5; i++) {
        memset(&channels[i], 0, sizeof(ledc_channel_config_t));
    }
    
    // Configurar parámetros específicos
    channels[0].gpio_num = SERVO1_PIN;
    channels[0].speed_mode = LEDC_LOW_SPEED_MODE;
    channels[0].channel = LEDC_CHANNEL_0;
    channels[0].timer_sel = LEDC_TIMER_0;

    channels[1].gpio_num = SERVO2_PIN;
    channels[1].speed_mode = LEDC_LOW_SPEED_MODE;
    channels[1].channel = LEDC_CHANNEL_1;
    channels[1].timer_sel = LEDC_TIMER_0;

    channels[2].gpio_num = SERVO3_PIN;
    channels[2].speed_mode = LEDC_LOW_SPEED_MODE;
    channels[2].channel = LEDC_CHANNEL_2;
    channels[2].timer_sel = LEDC_TIMER_0;

    channels[3].gpio_num = SERVO4_PIN;
    channels[3].speed_mode = LEDC_LOW_SPEED_MODE;
    channels[3].channel = LEDC_CHANNEL_3;
    channels[3].timer_sel = LEDC_TIMER_0;

    channels[4].gpio_num = SERVO5_PIN;
    channels[4].speed_mode = LEDC_LOW_SPEED_MODE;
    channels[4].channel = LEDC_CHANNEL_4;
    channels[4].timer_sel = LEDC_TIMER_0;

    for (int i = 0; i < 5; i++) {
        ESP_ERROR_CHECK(ledc_channel_config(&channels[i]));
    }

    ESP_LOGI(TAG, "5 servomotores configurados");
}

void controlar_servo1_canecas_llenas(bool alguna_caneca_llena) {
    if (alguna_caneca_llena && !servo1_activado) {
        mover_servomotor(LEDC_CHANNEL_SERVO1, 180);
        servo1_activado = true;
        ESP_LOGI(TAG, "Servo1 ACTIVADO por caneca llena");
    } else if (!alguna_caneca_llena && servo1_activado) {
        mover_servomotor(LEDC_CHANNEL_SERVO1, 0);
        servo1_activado = false;
        ESP_LOGI(TAG, "Servo1 DESACTIVADO - canecas vacias");
    }
}

void verificar_servo2_tiempo() {
    if (servo2_activado) {
        TickType_t tiempo_actual = xTaskGetTickCount();
        TickType_t tiempo_transcurrido = tiempo_actual - servo2_tiempo_activacion;
        
        if (tiempo_transcurrido >= pdMS_TO_TICKS(SERVO2_TIEMPO_ACTIVACION_MS)) {
            mover_servomotor(LEDC_CHANNEL_SERVO2, 0);
            servo2_activado = false;
            servo2_completado = true;
            ESP_LOGI(TAG, "Servo2 regresado a 0° - CICLO COMPLETADO");
            controlar_motor_despues_servo2();
        }
    }
}

void verificar_servo_nuevo_tiempo() {
    TickType_t tiempo_actual = xTaskGetTickCount();
    
    if (servo3_activado) {
        if ((tiempo_actual - servo_nuevo_tiempo_activacion) >= pdMS_TO_TICKS(SERVO_NUEVO_TIEMPO_MS)) {
            mover_servomotor(LEDC_CHANNEL_SERVO3, 0);
            servo3_activado = false;
            ESP_LOGI(TAG, "Servo3 (plásticos) regresado a 0°");
        }
    }
    
    if (servo4_activado) {
        if ((tiempo_actual - servo_nuevo_tiempo_activacion) >= pdMS_TO_TICKS(SERVO_NUEVO_TIEMPO_MS)) {
            mover_servomotor(LEDC_CHANNEL_SERVO4, 0);
            servo4_activado = false;
            ESP_LOGI(TAG, "Servo4 (papel/cartón) regresado a 0°");
        }
    }
    
    if (servo5_activado) {
        if ((tiempo_actual - servo_nuevo_tiempo_activacion) >= pdMS_TO_TICKS(SERVO_NUEVO_TIEMPO_MS)) {
            mover_servomotor(LEDC_CHANNEL_SERVO5, 0);
            servo5_activado = false;
            ESP_LOGI(TAG, "Servo5 (metales) regresado a 0°");
        }
    }
}

void setup_stepper_motor() {
    gpio_set_direction(EN_PIN, GPIO_MODE_OUTPUT);
    gpio_set_direction(STEP_PIN, GPIO_MODE_OUTPUT);
    gpio_set_direction(DIR_PIN, GPIO_MODE_OUTPUT);
    
    gpio_set_level(EN_PIN, 1);
    gpio_set_level(STEP_PIN, 0);
    gpio_set_level(DIR_PIN, 0);
    direccion_motor_fijada = true;
    
    ESP_LOGI(TAG, "Motor NEMA17 configurado");
}

void enable_stepper() {
    gpio_set_level(EN_PIN, 0);
    vTaskDelay(pdMS_TO_TICKS(10));
    ESP_LOGI(TAG, "Motor NEMA17 HABILITADO");
}

void disable_stepper() {
    gpio_set_level(EN_PIN, 1);
    ESP_LOGI(TAG, "Motor NEMA17 DESHABILITADO");
}

void controlar_motor_despues_servo2() {
    if (!motor_activado && motor_task_handle == NULL && servo2_completado) {
        ESP_LOGI(TAG, "Activando motor NEMA17 después de Servo2");
        
        BaseType_t result = xTaskCreate(
            [](void *pvParameters) {
                ESP_LOGI(TAG, "=== INICIANDO ROTACIÓN CONTINUA DEL MOTOR ===");
                enable_stepper();
                
                TickType_t start_time = xTaskGetTickCount();
                int step_counter = 0;
                
                while ((xTaskGetTickCount() - start_time) < pdMS_TO_TICKS(MOTOR_TIEMPO_ACTIVACION_MS)) {
                    gpio_set_level(STEP_PIN, 1);
                    esp_rom_delay_us(MOTOR_DELAY_US);
                    gpio_set_level(STEP_PIN, 0);
                    esp_rom_delay_us(MOTOR_DELAY_US);
                    step_counter++;
                    
                    if (step_counter % 200 == 0) {
                        ESP_LOGI(TAG, "Motor: %d pasos", step_counter);
                    }
                    
                    if (step_counter % 100 == 0) {
                        vTaskDelay(1);
                    }
                }
                
                disable_stepper();
                ESP_LOGI(TAG, "Rotación completada: %d pasos", step_counter);
                
                motor_activado = false;
                motor_task_handle = NULL;
                servo2_completado = false;
                vTaskDelete(NULL);
            }, 
            "motor_task", 4096, NULL, 1, &motor_task_handle
        );
        
        if (result == pdPASS) {
            motor_activado = true;
            motor_tiempo_activacion = xTaskGetTickCount();
        }
    }
}

void verificar_motor_tiempo() {
    if (motor_activado && motor_task_handle != NULL) {
        TickType_t tiempo_actual = xTaskGetTickCount();
        TickType_t tiempo_transcurrido = tiempo_actual - motor_tiempo_activacion;
        
        if (tiempo_transcurrido >= pdMS_TO_TICKS(15000)) {
            ESP_LOGW(TAG, "Motor timeout, forzando desactivación");
            disable_stepper();
            
            if (motor_task_handle != NULL) {
                vTaskDelete(motor_task_handle);
                motor_task_handle = NULL;
            }
            
            motor_activado = false;
            servo2_completado = false;
        }
    }
}

// Funciones display
void max7219_write_all(uint8_t reg, uint8_t *data) {
    uint8_t tx_data[MAX_DEVICES * 2];
    for (int i = 0; i < MAX_DEVICES; i++) {
        tx_data[i * 2] = reg;
        tx_data[i * 2 + 1] = data[i];
    }
    spi_transaction_t t;
    memset(&t, 0, sizeof(t));
    t.length = MAX_DEVICES * 16;
    t.tx_buffer = tx_data;
    spi_device_polling_transmit(spi, &t);
}

void max7219_write_all_same(uint8_t reg, uint8_t val) {
    uint8_t data[MAX_DEVICES];
    for (int i = 0; i < MAX_DEVICES; i++) data[i] = val;
    max7219_write_all(reg, data);
}

void max7219_init() {
    spi_bus_config_t buscfg;
    memset(&buscfg, 0, sizeof(buscfg));
    buscfg.miso_io_num = -1;
    buscfg.mosi_io_num = DIN_PIN;
    buscfg.sclk_io_num = CLK_PIN;
    buscfg.quadwp_io_num = -1;
    buscfg.quadhd_io_num = -1;
    buscfg.max_transfer_sz = MAX_DEVICES * 2;

    spi_device_interface_config_t devcfg;
    memset(&devcfg, 0, sizeof(devcfg));
    devcfg.clock_speed_hz = 1000000;
    devcfg.mode = 0;
    devcfg.spics_io_num = CS_PIN;
    devcfg.queue_size = 1;

    ESP_ERROR_CHECK(spi_bus_initialize(SPI_HOST_DEVICE, &buscfg, SPI_DMA_DISABLED));
    ESP_ERROR_CHECK(spi_bus_add_device(SPI_HOST_DEVICE, &devcfg, &spi));

    max7219_write_all_same(MAX7219_REG_SCANLIMIT, 7);
    max7219_write_all_same(MAX7219_REG_DECODEMODE, 0);
    max7219_write_all_same(MAX7219_REG_DISPLAYTEST, 0);
    max7219_write_all_same(MAX7219_REG_INTENSITY, 8);
    max7219_write_all_same(MAX7219_REG_SHUTDOWN, 1);
    
    ESP_LOGI(TAG, "MAX7219 inicializado");
}

void display_clear() {
    for (int i = 1; i <= 8; i++) {
        max7219_write_all_same(i, 0x00);
    }
    memset(display_buffer, 0, sizeof(display_buffer));
}

void display_refresh() {
    for (int row = 0; row < 8; row++) {
        uint8_t row_data[MAX_DEVICES];
        for (int dev = 0; dev < MAX_DEVICES; dev++) {
            uint8_t val = 0;
            for (int col = 0; col < 8; col++) {
                int buffer_col = dev * 8 + col;
                if (display_buffer[buffer_col] & (1 << row)) {
                    val |= (1 << (7 - col));
                }
            }
            row_data[dev] = val;
        }
        max7219_write_all(MAX7219_REG_DIGIT0 + row, row_data);
    }
}

void scroll_text_update() {
    memset(display_buffer, 0, sizeof(display_buffer));
    
    for (int i = 0; i < 32; i++) {
        int text_idx = (scroll_position + i) % (text_length * 6);
        int char_idx = text_idx / 6;
        int col_in_char = text_idx % 6;

        if (char_idx < text_length) {
            char current_char = scroll_text[char_idx];
            int font_idx = get_char_index(current_char);
            
            if (col_in_char < 5) {
                display_buffer[i] = font_5x7[font_idx][col_in_char];
            }
        }
    }

    display_refresh();
    
    scroll_position++;
    if (scroll_position >= text_length * 6) {
        scroll_position = 0;
    }
}

void setup_ultrasonic_pins() {
    gpio_set_direction(TRIG1_PIN, GPIO_MODE_OUTPUT);
    gpio_set_direction(TRIG2_PIN, GPIO_MODE_OUTPUT);
    gpio_set_direction(TRIG3_PIN, GPIO_MODE_OUTPUT);
    gpio_set_direction(TRIG4_PIN, GPIO_MODE_OUTPUT);
    
    gpio_set_direction(ECHO1_PIN, GPIO_MODE_INPUT);
    gpio_set_direction(ECHO2_PIN, GPIO_MODE_INPUT);
    gpio_set_direction(ECHO3_PIN, GPIO_MODE_INPUT);
    gpio_set_direction(ECHO4_PIN, GPIO_MODE_INPUT);
    
    gpio_set_level(TRIG1_PIN, 0);
    gpio_set_level(TRIG2_PIN, 0);
    gpio_set_level(TRIG3_PIN, 0);
    gpio_set_level(TRIG4_PIN, 0);
    
    ESP_LOGI(TAG, "Pines ultrasónicos configurados");
}

int read_ultrasonic(int trig_pin, int echo_pin) {
    gpio_set_level((gpio_num_t)trig_pin, 0);
    esp_rom_delay_us(2);
    gpio_set_level((gpio_num_t)trig_pin, 1);
    esp_rom_delay_us(10);
    gpio_set_level((gpio_num_t)trig_pin, 0);
    
    uint32_t timeout = 10000;
    uint32_t start_time = esp_timer_get_time();
    while (gpio_get_level((gpio_num_t)echo_pin) == 0) {
        if ((esp_timer_get_time() - start_time) > timeout) {
            return 999;
        }
    }
    
    start_time = esp_timer_get_time();
    while (gpio_get_level((gpio_num_t)echo_pin) == 1) {
        if ((esp_timer_get_time() - start_time) > 30000) {
            return 999;
        }
    }
    
    uint32_t echo_end = esp_timer_get_time();
    uint32_t duration = echo_end - start_time;
    
    int distance = (duration * 0.0343) / 2;
    
    if (distance < 2 || distance > 400) {
        return 999;
    }
    
    return distance;
}

void update_ultrasonic_display() {
    ultra1_value = read_ultrasonic(TRIG1_PIN, ECHO1_PIN);
    ultra2_value = read_ultrasonic(TRIG2_PIN, ECHO2_PIN);
    ultra3_value = read_ultrasonic(TRIG3_PIN, ECHO3_PIN);
    ultra4_value = read_ultrasonic(TRIG4_PIN, ECHO4_PIN);
    
    ESP_LOGI(TAG, "Sensores - U1:%dcm U2:%dcm U3:%dcm U4:%dcm", 
             ultra1_value, ultra2_value, ultra3_value, ultra4_value);
    
    bool objeto_detectado = (ultra1_value <= 10 && ultra1_value != 999);
    bool caneca1_llena = (ultra2_value <= 9 && ultra2_value != 999);
    bool caneca2_llena = (ultra3_value <= 9 && ultra3_value != 999);
    bool caneca3_llena = (ultra4_value <= 9 && ultra4_value != 999);
    bool alguna_caneca_llena = caneca1_llena || caneca2_llena || caneca3_llena;
    
    controlar_servo1_canecas_llenas(alguna_caneca_llena);
    
    // PRIORIDAD 1: Mostrar objeto detectado por ESP-NOW
    if (mostrar_objeto) {
        TickType_t tiempo_actual = xTaskGetTickCount();
        TickType_t tiempo_transcurrido = tiempo_actual - tiempo_display_objeto;
        
        if (tiempo_transcurrido < pdMS_TO_TICKS(TIEMPO_DISPLAY_OBJETO_MS)) {
            // Mostrar el objeto detectado
            snprintf(display_text, sizeof(display_text), "%s ", objeto_display);
        } else {
            // Tiempo terminado, volver al modo normal
            mostrar_objeto = false;
            objeto_detectado_espnow = false;
        }
    }
    // PRIORIDAD 2: Mostrar objeto detectado por ultrasonido
    else if (objeto_detectado_espnow) {
        snprintf(display_text, sizeof(display_text), "Obj: %s ", objeto_actual);
        objeto_detectado_espnow = false;
    }
    // PRIORIDAD 3: Mostrar objeto detectado por ultrasonido local
    else if (objeto_detectado) {
        strcpy(display_text, "OBJETO DETECTADO ");
    } 
    // PRIORIDAD 4: Mostrar canecas llenas
    else if (caneca1_llena) {
        strcpy(display_text, "CANECA 1 LLENA ");
    }
    else if (caneca2_llena) {
        strcpy(display_text, "CANECA 2 LLENA ");
    }
    else if (caneca3_llena) {
        strcpy(display_text, "CANECA 3 LLENA ");
    }
    // PRIORIDAD 5: Mensaje normal
    else {
        strcpy(display_text, "BIENVENIDO ");
    }
    
    strcpy(scroll_text, display_text);
    text_length = strlen(scroll_text);
}

extern "C" void app_main() {
    esp_err_t ret = nvs_flash_init();
    if (ret == ESP_ERR_NVS_NO_FREE_PAGES || ret == ESP_ERR_NVS_NEW_VERSION_FOUND) {
        ESP_ERROR_CHECK(nvs_flash_erase());
        ret = nvs_flash_init();
    }
    ESP_ERROR_CHECK(ret);
    
    gpio_set_direction(PINOFF, GPIO_MODE_OUTPUT);
    gpio_set_level(PINOFF, 1);
    
    gpio_set_direction(BUTTONOFF, GPIO_MODE_INPUT);
    gpio_set_pull_mode(BUTTONOFF, GPIO_PULLUP_ONLY);
    
    setup_ultrasonic_pins();
    setup_servomotores();
    setup_stepper_motor();
    setup_espnow();
    max7219_init();
    display_clear();

    strcpy(scroll_text, "Sistema reciclaje listo ");
    text_length = strlen(scroll_text);
    scroll_text_update();
    
    ESP_LOGI(TAG, "Sistema completamente inicializado");

    bool sistema_encendido = true;
    TickType_t last_scroll_time = xTaskGetTickCount();
    TickType_t last_sensor_time = xTaskGetTickCount();
    TickType_t last_servo_check_time = xTaskGetTickCount();
    const TickType_t scroll_delay = pdMS_TO_TICKS(50);
    const TickType_t sensor_delay = pdMS_TO_TICKS(500);
    const TickType_t servo_check_delay = pdMS_TO_TICKS(100);

    while (1) {
        int btnState = gpio_get_level(BUTTONOFF);

        if (btnState == 0) {
            ESP_LOGI(TAG, "Boton presionado, esperando...");
            int tiempo = 0;
            while (gpio_get_level(BUTTONOFF) == 0) {
                vTaskDelay(100 / portTICK_PERIOD_MS);
                tiempo += 100;
                if (tiempo >= 2000) {
                    sistema_encendido = !sistema_encendido;
                    if (sistema_encendido) {
                        gpio_set_level(PINOFF, 1);
                        max7219_write_all_same(MAX7219_REG_SHUTDOWN, 1);
                        strcpy(scroll_text, "Sistema ACTIVADO ");
                        text_length = strlen(scroll_text);
                        scroll_position = 0;
                        ESP_LOGI(TAG, "Sistema ENCENDIDO");
                    } else {
                        gpio_set_level(PINOFF, 0);
                        max7219_write_all_same(MAX7219_REG_SHUTDOWN, 0);
                        display_clear();
                        for (int i = 0; i < 5; i++) {
                            mover_servomotor((ledc_channel_t)i, 0);
                        }
                        disable_stepper();
                        
                        if (motor_task_handle != NULL) {
                            vTaskDelete(motor_task_handle);
                            motor_task_handle = NULL;
                        }
                        
                        servo1_activado = false;
                        servo2_activado = false;
                        servo3_activado = false;
                        servo4_activado = false;
                        servo5_activado = false;
                        motor_activado = false;
                        servo2_completado = false;
                        mostrar_objeto = false;
                        ESP_LOGI(TAG, "Sistema APAGADO");
                    }
                    vTaskDelay(500 / portTICK_PERIOD_MS);
                    break;
                }
            }
        }

        if (sistema_encendido) {
            if (xTaskGetTickCount() - last_sensor_time >= sensor_delay) {
                update_ultrasonic_display();
                last_sensor_time = xTaskGetTickCount();
            }
            
            if (xTaskGetTickCount() - last_servo_check_time >= servo_check_delay) {
                verificar_servo2_tiempo();
                verificar_servo_nuevo_tiempo();
                verificar_motor_tiempo();
                last_servo_check_time = xTaskGetTickCount();
            }
            
            if (xTaskGetTickCount() - last_scroll_time >= scroll_delay) {
                scroll_text_update();
                last_scroll_time = xTaskGetTickCount();
            }
        }

        vTaskDelay(50 / portTICK_PERIOD_MS);
    }
}
