<script>
export default {
  name: 'App',
  data() {
    return {
      plasticMaterials: [
        { name: 'Botellas', value: 78, icon: '🥤' },
        { name: 'Envases', value: 65, icon: '🛍️' },
        { name: 'Otros Plásticos', value: 42, icon: '🍶' }
      ],
      paperMaterials: [
        { name: 'Periódicos', value: 55, icon: '📰' },
        { name: 'Cajas', value: 82, icon: '📦' },
        { name: 'Otros Papeles', value: 38, icon: '📄' }
      ],
      aluminumMaterials: [
        { name: 'Latas', value: 91, icon: '🥫' },
        { name: 'Envases Metálicos', value: 47, icon: '🔩' },
        { name: 'Tapas', value: 29, icon: '🍾' }
      ],
      detectedObject: 'BOTELLA',
      motorStatus: 'INACTIVO',
      stats: {
        objectsToday: 47,
        accuracyRate: 94,
        kgRecycled: '12.8',
        activeTime: '6h 24m'
      },
      objectTypes: ['BOTELLA', 'LATA', 'PAPEL', 'CAJA', 'NO RECICLABLE']
    }
  },
  computed: {
    detectionClass() {
      return this.detectedObject === 'NO RECICLABLE' ? 'system-inactive' : 'object-detected';
    },
    motorClass() {
      return this.motorStatus === 'ACTIVO' ? 'system-active' : 'system-inactive';
    }
  },
  mounted() {
    // Iniciar actualizaciones en tiempo real
    this.startRealTimeUpdates();
  },
  methods: {
    startRealTimeUpdates() {
      // Actualizar valores cada 5 segundos
      setInterval(() => {
        this.updateMaterialValues();
        this.updateStats();
      }, 5000);
      
      // Cambiar objeto detectado cada 8 segundos
      setInterval(() => {
        this.changeObjectDetection();
      }, 8000);
      
      // Cambiar estado del motor cada 15 segundos
      setInterval(() => {
        this.toggleMotorStatus();
      }, 15000);
    },
    
    updateMaterialValues() {
      // Actualizar valores de plástico
      this.plasticMaterials = this.plasticMaterials.map(material => ({
        ...material,
        value: Math.min(100, Math.max(0, material.value + (Math.random() * 10 - 5)))
      }));
      
      // Actualizar valores de papel
      this.paperMaterials = this.paperMaterials.map(material => ({
        ...material,
        value: Math.min(100, Math.max(0, material.value + (Math.random() * 10 - 5)))
      }));
      
      // Actualizar valores de aluminio
      this.aluminumMaterials = this.aluminumMaterials.map(material => ({
        ...material,
        value: Math.min(100, Math.max(0, material.value + (Math.random() * 10 - 5)))
      }));
    },
    
    updateStats() {
      this.stats.objectsToday += Math.floor(Math.random() * 3);
      this.stats.kgRecycled = (parseFloat(this.stats.kgRecycled) + (Math.random() * 0.2)).toFixed(1);
      
      // Simular cambio en tiempo activo
      const timeParts = this.stats.activeTime.split(' ');
      let hours = parseInt(timeParts[0]);
      let minutes = parseInt(timeParts[1]);
      
      minutes += 1;
      if (minutes >= 60) {
        minutes = 0;
        hours += 1;
      }
      
      this.stats.activeTime = `${hours}h ${minutes}m`;
    },
    
    changeObjectDetection() {
      const randomIndex = Math.floor(Math.random() * this.objectTypes.length);
      this.detectedObject = this.objectTypes[randomIndex];
    },
    
    toggleMotorStatus() {
      this.motorStatus = this.motorStatus === 'ACTIVO' ? 'INACTIVO' : 'ACTIVO';
    }
  }
}
</script>
